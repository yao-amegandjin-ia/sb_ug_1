3// Columns (any order, extra columns are ignored):

26 public static List<MethodEntry> Parse(string csvText, ConfigReport report)
27 {
28 List<MethodEntry> methods = new List<MethodEntry>();
29 CsvTable table = CsvTable.Parse(csvText, FileName);
30
31 if (!table.HasColumn(IdColumn) || !table.HasColumn(NameColumn))
32 {
33 report.Add(FileName, "the first row must have an \"ID\" column and a \"Name\" column.");
34 return methods;
35 }
36
37 HashSet<int> usedIds = new HashSet<int>();
38 HashSet<string> usedNames = new HashSet<string>();
39
40 for (int r = 0; r < table.RowCount; r++)
41 {
42 int row = table.SpreadsheetRow(r);
43 string idText = table.Get(r, IdColumn);
44 string name = table.Get(r, NameColumn);
45
46 if (!int.TryParse(idText, out int id) || id <= 0)
47 {
48 report.Add(FileName, row, $"ID \"{idText}\" must be a whole number of 1 or more. Row skipped.");
49 continue;
50 }
51 if (!usedIds.Add(id))
52 {
53 report.Add(FileName, row, $"ID {id} is already used by another methodology. Row skipped.");
54 continue;
55 }
56 if (name.Length == 0)
57 {
58 report.Add(FileName, row, $"methodology {id} has no name. Row skipped.");
59 continue;
60 }
61 if (!usedNames.Add(CsvTable.Normalize(name)))
62 report.Add(FileName, row, $"the name \"{name}\" is used twice. Requests will only match the first one.");
63
64 string description = table.Get(r, DescriptionColumn);
65 if (description.Length == 0)
66 report.Add(FileName, row, $"\"{name}\" has no description (it still loads).");
67
68 methods.Add(new MethodEntry
69 {
70 id = id,
71 name = name,
72 description = description,
73 aliases = CsvTable.SplitList(table.Get(r, AliasColumn))
74 });
75 }
76
77 if (methods.Count == 0)
78 report.Add(FileName, "no methodologies were loaded.");
79
80 // sorted by ID so lists in the UI don't depend on row order
81 methods.Sort((a, b) => a.id.CompareTo(b.id));
82 return methods;
83 }

91 public static MethodEntry Find(IEnumerable<MethodEntry> methods, string typed)
92 {
93 string key = CsvTable.Normalize(typed);
94 if (key.Length == 0 || methods == null) return null;
95
96 bool isNumber = int.TryParse(typed.Trim(), out int id);
97 string shortKey = WithoutResearchWord(key);
98 MethodEntry looseMatch = null;
99
100 foreach (MethodEntry m in methods)
101 {
102 if (isNumber && m.id == id) return m;
103 if (CsvTable.Normalize(m.name) == key) return m;
104 if (m.aliases != null)
105 foreach (string alias in m.aliases)
106 if (CsvTable.Normalize(alias) == key) return m;
107
108 // "Field" for "Field Research", or "Systematic Review" for "Systematic Reviews"
109 if (looseMatch == null && WithoutResearchWord(CsvTable.Normalize(m.name)) == shortKey)
110 looseMatch = m;
111 }
112 return looseMatch;
113 }
114
115 // "fieldresearch" -> "field", "systematicreviews" -> "systematicreview"
116 private static string WithoutResearchWord(string normalized)
117 {
118 string s = normalized;
119 if (s.EndsWith("research") && s.Length > "research".Length) s = s.Substring(0, s.Length - "research".Length);
120 if (s.EndsWith("s") && s.Length > 1) s = s.Substring(0, s.Length - 1);
121 return s;
122 }
123 }
