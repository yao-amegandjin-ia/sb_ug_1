141 public bool TryRecordStudy(int requestId, ResearchApproach approach, int methodologyId, int reward)
142 {
143 if (reward < 0) throw new ArgumentOutOfRangeException(nameof(reward));
144 if (methodologyId < 0) throw new ArgumentOutOfRangeException(nameof(methodologyId));
145
146 // the approach is saved whether or not this pays, the same as the methodology below
147 state.Player.StudyApproaches[requestId] = (int)approach;
148
149 // -1 is a reserved/unanswered request, not a final answer.
150 if (state.Player.StudyResults.TryGetValue(requestId, out int method) && method != -1)
151 {
152 // Keep the latest choice without paying the same request twice.
153 state.Player.StudyResults[requestId] = methodologyId;
154 return false;
155 }
156 int money = checked(state.Player.CurrentMoney + reward);
157 int total = checked(state.Player.TotalMoneyEarned + reward);
158 state.Player.StudyResults[requestId] = methodologyId;
159 state.Player.CurrentMoney = money;
160 state.Player.TotalMoneyEarned = total;
161 state.Clients.RemoveAll(c => c.RequestId == requestId);
162 MoneyEarned?.Invoke(reward);
163 return true;
164 }
