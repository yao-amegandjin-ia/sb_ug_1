2// The first row is the header row. Columns are found by their header name, so the person writing the
3// file can put the columns in any order and add their own extra columns (like "Notes") without breaking anything.
4// Plain C#, no Unity objects, so it can be used anywhere (and tested outside Unity).

66 public static CsvTable Parse(string text, string fileName)
67 {
68 CsvTable table = new CsvTable(fileName);
69 if (string.IsNullOrEmpty(text)) return table;
70
71 // Excel's "CSV UTF-8" option puts an invisible marker (a BOM) at the very start. Remove it,
72 // or the first header would be "﻿ID" and the ID column would never be found.
73 if (text[0] == '﻿') text = text.Substring(1);
74
75 char delimiter = DetectDelimiter(text);
76 List<string[]> records = SplitRecords(text, delimiter);
77 if (records.Count == 0) return table;
78
79 // header row
80 string[] header = records[0];
81 for (int i = 0; i < header.Length; i++)
82 {
83 string key = Normalize(header[i]);
84 if (key.Length > 0 && !table.columnIndex.ContainsKey(key))
85 table.columnIndex[key] = i;
86 }
87
88 // data rows. record index 0 is spreadsheet row 1, so record r is row r + 1
89 for (int r = 1; r < records.Count; r++)
90 {
91 string[] cells = records[r];
92 if (IsBlank(cells)) continue; // empty rows spreadsheets leave at the end
93 if (cells[0].TrimStart().StartsWith("#")) continue; // "# ..." rows are comments for the writer
94 table.rows.Add(cells);
95 table.rowNumbers.Add(r + 1);
96 }
97
98 return table;
99 }

105 public static string Normalize(string text)
106 {
107 if (string.IsNullOrEmpty(text)) return "";
108 StringBuilder sb = new StringBuilder(text.Length);
109 foreach (char c in text)
110 {
111 if (char.IsLetterOrDigit(c))
112 sb.Append(char.ToLowerInvariant(c));

134 // Excel in some countries saves with ; or tabs instead of commas. Look at the header row
135 // (outside quotes) and use whichever of the three appears most. Commas win a tie.
136 private static char DetectDelimiter(string text)
137 {
138 int commas = 0, semicolons = 0, tabs = 0;
139 bool inQuotes = false;
140 foreach (char c in text)
141 {
142 if (c == '"') inQuotes = !inQuotes;
143 else if (!inQuotes)
144 {
145 if (c == '\n' || c == '\r') break; // only the first line
146 if (c == ',') commas++;
147 else if (c == ';') semicolons++;
148 else if (c == '\t') tabs++;
149 }
150 }
151 if (tabs > commas && tabs >= semicolons) return '\t';
152 if (semicolons > commas) return ';';
153 return ',';
154 }

156 // The CSV rules spreadsheets use:
157 // - cells are separated by the delimiter, rows by a line break
158 // - a cell with a comma, quote or line break in it is wrapped in "double quotes"
159 // - a quote inside a quoted cell is written twice: "She said ""hi"""
160 private static List<string[]> SplitRecords(string text, char delimiter)
161 {
162 List<string[]> records = new List<string[]>();
163 List<string> cells = new List<string>();
164 StringBuilder cell = new StringBuilder();
165 bool inQuotes = false;
166
167 for (int i = 0; i < text.Length; i++)
168 {
169 char c = text[i];
170
171 if (inQuotes)
172 {
173 if (c == '"')
174 {
175 if (i + 1 < text.Length && text[i + 1] == '"') { cell.Append('"'); i++; } // "" -> "
176 else inQuotes = false; // closing quote
177 }
178 else cell.Append(c); // anything goes inside quotes, including commas and line breaks
179 continue;
180 }
181
182 if (c == '"') inQuotes = true;
183 else if (c == delimiter) { cells.Add(cell.ToString()); cell.Clear(); }
184 else if (c == '\r' || c == '\n')
185 {
186 if (c == '\r' && i + 1 < text.Length && text[i + 1] == '\n') i++; // Windows line break \r\n
187 cells.Add(cell.ToString()); cell.Clear();
188 records.Add(cells.ToArray()); cells.Clear();
189 }
190 else cell.Append(c);
191 }
192
193 // last row (files often don't end with a line break)
194 if (cell.Length > 0 || cells.Count > 0)
195 {
196 cells.Add(cell.ToString());
197 records.Add(cells.ToArray());
198 }
199 return records;
200 }
