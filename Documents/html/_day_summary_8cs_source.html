1// DaySummary - shows up when the day ends (FirmController.DayEnded), or when something calls Show(day).
2// Continue asks FirmController for the next day.
3// top: how many of each result today. middle: every client from today, what the player answered
4// (approach + methodology), what it paid, the best answer and why (the Rationale column of the requests
5// spreadsheet). bottom: money and Continue.
6// it keeps its own list of today's clients from FirmController.ClientResolved. that event carries the request,
7// the methodology picked and the outcome, so nothing has to be looked up in the saved study results.

233 string YourAnswer(ClientResult entry)
234 {
235 switch (entry.Outcome)
236 {
237 case ClientOutcome.Lost: return "They left before you helped them";
238 case ClientOutcome.CorrectReject:
239 case ClientOutcome.WrongReject: return "We can't help";
240 default:
241 // e.g. "Qualitative: Field Research", the same shape as the best answer below
242 string method = MethodLibrary.GetName(entry.MethodologyId);
243 return entry.Approach != ResearchApproach.None ? entry.Approach + ": " + method : method;
244 }
