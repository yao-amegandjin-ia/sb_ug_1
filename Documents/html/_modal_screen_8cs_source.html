1// ModalScreen - base class for the full-screen popups (request, referral, method match, hire).
2// each popup sits on its own GameObject with a PanelRenderer:
3// Panel Settings = ModalPanelSettings, Source Asset = that screen's UXML.
4// a PanelRenderer builds its UI on its own schedule and tells us through the reload callback,
5// so the Q<>() lookups happen in BindUI instead of Start. the callback runs again if the UI is rebuilt
6// (e.g. you swap the Source Asset or Panel Settings), and BindUI just looks everything up again.
7// screens only show things and pass clicks on to the FirmController, which has the game rules.
8// keep these GameObjects active. the screen hides itself with display: none, it doesn't disable anything.
9

115 // the version only changes when the UI is really rebuilt, so skip repeat calls.
116 // otherwise every button would get its click handler added twice
117 private void OnUIReload(PanelRenderer renderer, VisualElement root, int version)
118 {
119 if (root == Root && version == boundVersion) return;
120 Root = root;
121 boundVersion = version;
122
123 SetUpOverlay(root);
124 BindUI(root);
125
126 ApplyVisibility();
127 if (IsOpen) OnShown();
128 }

130 // cover the whole screen and center the card. a side panel places itself with position: absolute in its USS.
131 // when the lab isn't blocked the overlay is see-through and ignores clicks, so only the panel catches them
132 private void SetUpOverlay(VisualElement root)
133 {
134 root.style.position = Position.Absolute;
135 root.style.left = 0;
136 root.style.right = 0;
137 root.style.top = 0;
138 root.style.bottom = 0;
139 root.style.justifyContent = Justify.Center;
140 root.style.alignItems = Align.Center;
141 root.style.backgroundColor = blockLabBehind ? dimColor : Color.clear;
142 root.pickingMode = blockLabBehind ? PickingMode.Position : PickingMode.Ignore;
143
144 if (extraStyleSheets == null) return;
145 foreach (StyleSheet sheet in extraStyleSheets)
146 {
147 if (sheet != null && !root.styleSheets.Contains(sheet))
148 root.styleSheets.Add(sheet);
149 }
150 }
