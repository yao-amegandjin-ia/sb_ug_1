69 public void Generate()
70 {
71 if (!ValidateSettings()) return;
72
73 researchers.Clear();
74
75 if (overrideSeed)
76 GameSeed.Set(seedOverride);
77
78 Random rng = GameSeed.CreateRandom(RandomStreamId);
79 Debug.Log($"ResearcherGenerator: generating with seed {GameSeed.Seed} (override {(overrideSeed ? "ON" : "OFF")}).");
80
81 //Unique heads: shuffle all head indices, then deal one to each researcher.
82 List<int> heads = new List<int>();
83 for (int h = 0; h < headCount; h++) heads.Add(h);
84 Shuffle(heads, rng);
85
86 //Pool of methodologies not yet handed out in the current cycle.
87 List<int> unused = new List<int>();
88 RefillPool(unused, rng);
89
90 HashSet<string> usedNames = new HashSet<string>();
91
92 for (int i = 0; i < researcherCount; i++)
93 {
94 string fullName = GenerateUniqueName(rng, usedNames);
95 Vector3Int look = new Vector3Int(heads[i], rng.Next(bodyCount), rng.Next(legCount));
96
97 int experience = 0;
98 int speed = NextInclusive(rng, speedRange);
99 int quality = NextInclusive(rng, qualityRange);
100
101 List<int> methods = DrawMethodologies(unused, rng);
102
103 // Sequential IDs: 1 through researcherCount.
104 int id = i + 1;
105
106 ResearcherData r = new ResearcherData();
107 r.Init(id, fullName, look, experience, speed, quality,methods);
108 researchers.Add(r);
109
110 Debug.Log($"ResearcherGenerator: Researcher generated: " + r.Name);
111 }

170 private List<int> DrawMethodologies(List<int> unused, Random rng)
171 {
172
173 List<int> unlocked = new List<int>();
174
175 while (unlocked.Count < 4)
176 {
177 // Start a new cycle (every methodology has now been used equally often).
178 if (unused.Count == 0)
179 RefillPool(unused, rng);
180
181 //finds the first index in unused, where it also is not in unlocked
182 int index = unused.FindIndex(m => !unlocked.Contains(m));
183
184 // Only duplicates of this researcher's methodologies are left; start a new cycle.
185 if (index < 0)
186 {
187 RefillPool(unused, rng);
188 index = unused.FindIndex(m => !unlocked.Contains(m));
189 }
190
191 unlocked.Add(unused[index]);
192 unused.RemoveAt(index);
193 }
194
195 return unlocked;
196 }
