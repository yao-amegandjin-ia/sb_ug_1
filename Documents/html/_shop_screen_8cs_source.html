2// one window with a tab button for every ShopTab component on this same GameObject (Upgrades, Hire, ...).
3// it always opens on the first tab. buying or hiring doesn't close it, the tab just rebuilds.
4// the player leaves with the CLOSE button. for now the HUD's Upgrades button opens it.

149 private void Rebuild(bool keepScroll)
150 {
151 if (moneyText != null) moneyText.text = "$" + (dataManager != null ? dataManager.PlayerData.CurrentMoney : 0).ToString("N0");
152 if (list == null || tabs.Length == 0) return;
153
154 Vector2 scroll = list.scrollOffset;
155 list.Clear();
156 tabs[currentTab].Fill(list);
157
158 // the new cards haven't been laid out yet, so put the scroll back on the next frame
159 if (keepScroll) list.schedule.Execute(() => list.scrollOffset = scroll);
160 else list.scrollOffset = Vector2.zero;
161 }
