1// GameUIPanel - goes on the GameUI object, next to its PanelRenderer
2// (Source Asset = GameUI.uxml, Panel Settings = GameUIPanelSettings).
3// the start screen, HUD, codex, manage firm and day summary all live in that one UXML.
4// a PanelRenderer builds its UI on its own schedule and tells us through the reload callback,
5// so the lookups happen there instead of in Start. this hands the root to every IGameUIPart on this object.
6// the callback runs again if the UI is rebuilt (e.g. you swap the Source Asset), and every part just looks everything up again.
7

39 void OnUIReload(PanelRenderer renderer, VisualElement root, int version)
40 {
41 // same UI as last time, so skip it. otherwise every button would get its click handler twice
42 if (root == boundRoot && version == boundVersion) return;
43 boundRoot = root;
44 boundVersion = version;
45
46 // fill the screen, but let clicks through to the lab wherever there's no actual panel or button
47 root.style.position = Position.Absolute;
48 root.style.left = 0;
49 root.style.right = 0;
50 root.style.top = 0;
51 root.style.bottom = 0;
52 root.pickingMode = PickingMode.Ignore;
53
54 foreach (IGameUIPart part in GetComponents<IGameUIPart>())
55 part.Bind(root);
56 }
57 }
