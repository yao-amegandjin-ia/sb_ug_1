1// FirmController - the game rules for what the player decides about clients and researchers.
2// goes on the same GameObject as GameInitializer in the Bootstrap scene.
3// the screens only show things and pass on clicks. every click that changes the game calls one function here,
4// so the rules can be read (and changed) in one place without any UI code around them.
5// money and study results live in the DataManager. the HUD and DaySummary hear what happened through the events below.
6// it also owns the day: the timer, which day it is, starting the next one and ending this one.
7// the HUD only shows TimeLeft / CurrentDay, and DaySummary opens on DayEnded and calls NextDay.

231 public bool RejectRequest(Client client)
232 {
233 if (!client) return false;
234
235 RequestData request = client.Data.Request;
236 bool correct = StudyScoring.IsCorrectReject(request);
237 int reward = correct ? correctRejectReward : 0;
238
239 if (request != null)
240 RecordStudy(request.RequestId, ResearchApproach.None, MethodLibrary.NoneId, reward);
241 Debug.Log($"{client.Data.Name} denied: {(correct ? "correct" : "wrong, this one was researchable")}, +${reward}");
242
243 client.Data.Complete(); // "request was rejected"
244 client.Leave();
245
246 RequestRejected?.Invoke(correct);
247 ClientResolved?.Invoke(new ClientResult
248 {
249 Request = request,
250 MethodologyId = MethodLibrary.NoneId,
251 Outcome = correct ? ClientOutcome.CorrectReject : ClientOutcome.WrongReject,
252 Reward = reward
253 });
254 return correct;
255 }

384 public MatchGrade CompleteStudy(Client client, Researcher researcher, ResearchApproach approach, int methodologyId)
385 {
386 if (!client || !researcher) return new MatchGrade(false, StudyOutcome.Wrong);
387
388 RequestData request = client.Data.Request;
389 MatchGrade grade = StudyScoring.Grade(request, approach, methodologyId); // methodology IDs come from the methodologies spreadsheet
390 int reward = matchRewards.For(grade);
391
392 if (request != null)
393 RecordStudy(request.RequestId, approach, methodologyId, reward);
394 Debug.Log($"{client.Data.Name} matched with {researcher.Data.Name}: {approach} / {MethodLibrary.GetName(methodologyId)} ({grade}): +${reward}");
395
396 client.Data.AssignTo(researcher.Data.Id, methodologyId);
397 client.Data.Complete();
398 client.Leave();
399
400 StudyFinished?.Invoke(grade.Method);
401 ClientResolved?.Invoke(new ClientResult
402 {
403 Request = request,
404 Approach = approach,
405 ApproachCorrect = grade.ApproachCorrect,
406 MethodologyId = methodologyId,
407 Outcome = ClientResult.FromStudy(grade.Method),
408 Reward = reward
409 });
410 return grade;
411 }
