136 public void Open(QuizType quizType)
137 {
138 if (panel == null)
139 {
140 Debug.LogWarning("QuizScreen: the UI hasn't loaded yet.");
141 return;
142 }
143 if (allQuestions == null) allQuestions = LoadQuestions();
144
145 questions = new List<QuizQuestion>();
146 foreach (QuizQuestion q in allQuestions)
147 if (q.quiz == quizType) questions.Add(q);
148
149 if (questions.Count == 0)
150 {
151 Debug.LogWarning("QuizScreen: no " + quizType + " questions in " + FileName);
152 return;
153 }
154
155 type = quizType;
156 Shuffle(questions, GameSeed.CreateRandom(QuizStreamId, (int)type));
157 answers.Clear();
158 current = 0;
159 showingResults = false;
160
161 // both of these use the left side and would sit behind the quiz
162 CodexPanel codex = GetComponent<CodexPanel>();
163 if (codex != null) codex.Close();
164 ManageFirmPanel firm = GetComponent<ManageFirmPanel>();
165 if (firm != null) firm.Close();
166
167 title.text = type == QuizType.MidGame ? "QUICK CHECK-IN" : "FINAL QUIZ";
168 questionStart = Time.unscaledTime;
169 ShowQuestion();
170 panel.RemoveFromClassList("hidden");
171 }

173 // reads the spreadsheet. columns: ID, Quiz (Mid or Post), Question, A, B, C, D, Answer (the letter)
174 List<QuizQuestion> LoadQuestions()
175 {
176 List<QuizQuestion> loaded = new List<QuizQuestion>();
177 ConfigReport report = new ConfigReport();
178
179 string text = ConfigFiles.Read(FileName, report);
180 if (text != null)
181 {
182 CsvTable table = CsvTable.Parse(text, FileName);
183 for (int row = 0; row < table.RowCount; row++)
184 {
185 QuizQuestion q = ReadRow(table, row, report);
186 if (q != null) loaded.Add(q);
187 }
188 }
189
190 ConfigFiles.Log(report);
191 return loaded;
192 }

194 // one row of the spreadsheet, or null (with a report entry) if something in it is wrong
195 QuizQuestion ReadRow(CsvTable table, int row, ConfigReport report)
196 {
197 int line = table.SpreadsheetRow(row);
198 QuizQuestion q = new QuizQuestion();
199
200 if (!int.TryParse(table.Get(row, "ID"), out q.id))
201 {
202 report.Add(FileName, line, "the ID isn't a number");
203 return null;
204 }
205
206 string quiz = table.Get(row, "Quiz").ToLower();
207 if (quiz.StartsWith("mid")) q.quiz = QuizType.MidGame;
208 else if (quiz.StartsWith("post")) q.quiz = QuizType.PostGame;
209 else
210 {
211 report.Add(FileName, line, "Quiz should be Mid or Post");
212 return null;
213 }
214
215 q.question = table.Get(row, "Question");
216 if (q.question == "")
217 {
218 report.Add(FileName, line, "the question is empty");
219 return null;
220 }
221
222 // empty option cells are skipped, so a question can have 2 or 3 options too
223 string[] letters = { "A", "B", "C", "D" };
224 int answerIndex = -1;
225 string answer = table.Get(row, "Answer").ToUpper();
226 foreach (string letter in letters)
227 {
228 string option = table.Get(row, letter);
229 if (option == "") continue;
230 if (letter == answer) answerIndex = q.options.Count;
231 q.options.Add(option);
232 }
233
234 if (q.options.Count < 2)
235 {
236 report.Add(FileName, line, "needs at least 2 options");
237 return null;
238 }
239 if (answerIndex < 0)
240 {
241 report.Add(FileName, line, "Answer should be the letter of one of the options (A, B, C or D)");
242 return null;
243 }
244 q.answer = answerIndex;
245 return q;
246 }

248 // Fisher-Yates
249 void Shuffle(List<QuizQuestion> items, System.Random rng)
250 {
251 for (int i = items.Count - 1; i > 0; i--)
252 {
253 int j = rng.Next(i + 1);
254 QuizQuestion temp = items[i];
255 items[i] = items[j];
256 items[j] = temp;
257 }
258 }

300 void FinishQuiz()
301 {
302 current = -1;
303 showingResults = true;
304
305 Debug.Log(type + " quiz done: " + CorrectCount() + "/" + answers.Count + " right in " + FormatTime(TotalTime()));
306 foreach (QuizAnswer a in answers)
307 Debug.Log(" question " + a.question.id + ": picked " + Letter(a.picked) + (a.IsCorrect() ? " (right)" : " (wrong)") + ", " + a.seconds.ToString("0.0") + "s");
308
309 int earned = CorrectCount() * correctReward;
310 if (earned > 0)
311 {
312 FirmController firm = FindAnyObjectByType<FirmController>();
313 if (firm != null) firm.AddMoney(earned);
314 else Debug.LogWarning("QuizScreen: no FirmController found, so the quiz money wasn't added. Is the Research_Lab scene loaded?");
315 }
