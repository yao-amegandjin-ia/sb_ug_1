1// RequestConfig - turns requests.csv into the RequestData list the ClientGenerator deals clients from.
2//
3// Columns (any order, extra columns like "Notes" are ignored):
4// ID a whole number, 1 or more, never reused. Saves remember requests by this number.
5// Client Name who brings the request
6// Problem what the client says
7// Researchable yes / no. Leave it blank and it's worked out: a request with a Best Method is researchable
8// Approach Quantitative, Qualitative or Mixed (step one of matching)
9// Best Method the methodology name from methodologies.csv that scores best (step two)
10// Good Methods other methodology names that get partial credit, separated by ; or ,

34 public static List<RequestData> Parse(string csvText, IReadOnlyList<MethodEntry> methods, ConfigReport report)
35 {
36 List<RequestData> requests = new List<RequestData>();
37 CsvTable table = CsvTable.Parse(csvText, FileName);
38
39 if (!table.HasColumn(IdColumn) || !table.HasColumn(ProblemColumn))
40 {
41 report.Add(FileName, "the first row must have an \"ID\" column and a \"Problem\" column.");
42 return requests;
43 }
44 foreach (string[] optional in new[] { NameColumn, ApproachColumn, BestColumn, GoodColumn, RationaleColumn })
45 {
46 if (!table.HasColumn(optional))
47 report.Add(FileName, $"there is no \"{optional[0]}\" column, so it's blank for every request.");
48 }
49
50 HashSet<int> usedIds = new HashSet<int>();
51
52 for (int r = 0; r < table.RowCount; r++)
53 {
54 int row = table.SpreadsheetRow(r);
55 RequestData request = ParseRow(table, r, row, methods, report);
56 if (request == null) continue;
57
58 if (!usedIds.Add(request.RequestId))
59 {
60 report.Add(FileName, row, $"ID {request.RequestId} is already used by another request. Row skipped.");
61 continue;
62 }
63 requests.Add(request);
64 }
65
66 if (requests.Count == 0)
67 report.Add(FileName, "no requests were loaded.");
68 return requests;
69 }

71 // one row -> one request, or null if the row can't be used at all
72 private static RequestData ParseRow(CsvTable table, int r, int row, IReadOnlyList<MethodEntry> methods, ConfigReport report)
73 {
74 string idText = table.Get(r, IdColumn);
75 if (!int.TryParse(idText, out int id) || id <= 0)
76 {
77 report.Add(FileName, row, $"ID \"{idText}\" must be a whole number of 1 or more. Row skipped.");
78 return null;
79 }
80
81 string problem = table.Get(r, ProblemColumn);
82 if (problem.Length == 0)
83 {
84 report.Add(FileName, row, $"request {id} has no Problem text. Row skipped.");
85 return null;
86 }
87
88 RequestData request = new RequestData
89 {
90 RequestId = id,
91 Name = table.Get(r, NameColumn),
92 ProblemText = problem,
93 Rationale = table.Get(r, RationaleColumn)
94 };
95 if (request.Name.Length == 0)
96 report.Add(FileName, row, $"request {id} has no Client Name (it still loads).");
97
98 // the answers
99 request.Approach = ParseApproach(table.Get(r, ApproachColumn), row, report);
100 request.BestMethodologyId = LookUpMethod(table.Get(r, BestColumn), methods, row, report);
101 foreach (string name in CsvTable.SplitList(table.Get(r, GoodColumn)))
102 {
103 int goodId = LookUpMethod(name, methods, row, report);
104 if (goodId > 0 && goodId != request.BestMethodologyId && !request.GoodMethodologyIds.Contains(goodId))
105 request.GoodMethodologyIds.Add(goodId);
106 }
107
108 bool hasAnswer = request.BestMethodologyId > 0;
109 request.IsResearchable = ParseResearchable(table.Get(r, ResearchableColumn), hasAnswer, row, report);
110
111 if (request.IsResearchable)
112 {
113 if (!hasAnswer)
114 report.Add(FileName, row, $"request {id} is researchable but has no Best Method, so nothing can score best.");
115 if (request.Approach == ResearchApproach.None)
116 report.Add(FileName, row, $"request {id} is researchable but has no Approach (Quantitative, Qualitative or Mixed).");
117 if (request.Rationale.Length == 0)
118 report.Add(FileName, row, $"request {id} has no Rationale, so the day summary can't explain the answer.");
119 }
120 else if (hasAnswer || request.Approach != ResearchApproach.None || request.GoodMethodologyIds.Count > 0)
121 {
122 // "can't be researched" wins, so the answers are cleared to keep the data consistent
123 report.Add(FileName, row, $"request {id} counts as not researchable (Researchable is \"no\", or blank with no Best Method), " +
124 "so its Approach and methods are ignored.");
125 request.Approach = ResearchApproach.None;
126 request.BestMethodologyId = 0;
127 request.GoodMethodologyIds.Clear();
128 }
129
130 return request;
131 }

133 private static ResearchApproach ParseApproach(string text, int row, ConfigReport report)
134 {
135 switch (CsvTable.Normalize(text))
136 {
137 case "": return ResearchApproach.None;
138 case "quantitative": case "quantitativeresearch": case "quant":
139 return ResearchApproach.Quantitative;
140 case "qualitative": case "qualitativeresearch": case "qual":
141 return ResearchApproach.Qualitative;
142 case "mixed": case "mixedmethod": case "mixedmethods": case "mixedmethodsresearch": case "both":
143 return ResearchApproach.Mixed;
144 default:
145 report.Add(FileName, row, $"Approach \"{text}\" isn't one of Quantitative, Qualitative or Mixed. Left blank.");
146 return ResearchApproach.None;
147 }
148 }

150 private static bool ParseResearchable(string text, bool hasAnswer, int row, ConfigReport report)
151 {
152 switch (CsvTable.Normalize(text))
153 {
154 case "": return hasAnswer; // blank: researchable if a Best Method was given
155 case "yes": case "y": case "true": case "1": case "x":
156 return true;
157 case "no": case "n": case "false": case "0":
158 return false;
159 default:
160 report.Add(FileName, row, $"Researchable \"{text}\" should be yes or no. Treated as {(hasAnswer ? "yes" : "no")}.");
161 return hasAnswer;
162 }
163 }

165 // methodology name (or ID) -> ID. 0 if blank or unknown
166 private static int LookUpMethod(string typed, IReadOnlyList<MethodEntry> methods, int row, ConfigReport report)
167 {
168 if (string.IsNullOrWhiteSpace(typed)) return 0;
169
170 MethodEntry method = MethodologyConfig.Find(methods, typed);
171 if (method != null) return method.id;
172
173 report.Add(FileName, row, $"\"{typed}\" isn't in {MethodologyConfig.FileName}. " +
174 $"Check the spelling, or add it there. Known methods: {ListNames(methods)}.");
175 return 0;
176 }

178 private static string ListNames(IReadOnlyList<MethodEntry> methods)
179 {
180 if (methods == null || methods.Count == 0) return "(none loaded)";
181 List<string> names = new List<string>();
182 foreach (MethodEntry m in methods) names.Add(m.name);
183 return string.Join(", ", names);
