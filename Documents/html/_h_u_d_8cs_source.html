1// HUD - goes on the GameUI object. bottom bar: codex, manage firm, studies, money, day, timer.
2// adding this script also adds the other UI scripts it needs (RequireComponent).
3// other scripts should call AddMoney / AddStudyResult / SetOngoing instead of touching the labels.
4// study results and correct rejects come from FirmController's events, so the screens don't have to tell the HUD.
5// the day timer lives in FirmController. the HUD only shows it: Update reads FirmController.TimeLeft,
6// and DayStarted resets the "today" counts.
7// the labels are looked up in Bind (called by GameUIPanel when the PanelRenderer has built the UI),
8// the game systems are found in Start. either one can run first.

114 // listens to the firm's events. safe to call twice: it lets go of the old firm first, so nothing is counted double
115 void Connect(FirmController firm)
116 {
117 if (firm == firmRules && firm != null) return;
118 Disconnect();
119 firmRules = firm;
120 if (firmRules == null) return;
121
122 firmRules.MoneyEarned += OnMoneyEarned;
123 firmRules.MoneySpent += OnMoneySpent;
124 firmRules.StudyFinished += OnStudyFinished;
125 firmRules.RequestRejected += OnRequestRejected;
126 firmRules.ClientLost += OnClientLost;
127 firmRules.DayStarted += OnDayStarted;
128 firmRules.DayEnded += OnDayEnded;
129 Refresh();
130 }
131
132 void Disconnect()

139 firmRules.ClientLost -= OnClientLost;
140 firmRules.DayStarted -= OnDayStarted;
141 firmRules.DayEnded -= OnDayEnded;
