2// goes on the same GameObject as GameInitializer and FirmController in the Bootstrap scene.
3// it only knows the upgrade assets in its list and how many levels of each the player owns.
4// the scripts that use a bonus ask for it (HUD for day time, ClientGenerator for patience, ...),

90 public bool TryBuy(UpgradeDefinition upgrade)
91 {
92 if (!CanBuy(upgrade)) return false;
93
94 int cost = GetNextCost(upgrade);
95 if (!firmController.TrySpend(cost)) return false;
96
97 int newLevel = GetLevel(upgrade) + 1;
98 SetLevel(upgrade.Id, newLevel);
99 Debug.Log($"Bought {upgrade.DisplayName} level {newLevel}/{upgrade.MaxLevel} for ${cost} (${dataManager.PlayerData.CurrentMoney} left)");
100
101 UpgradePurchased?.Invoke(upgrade, newLevel);
102 UpgradesChanged?.Invoke();
103 return true;
104 }

143 public void LoadSaveData(Dictionary<string, int> data)
144 {
145 owned.Clear();
146 if (data != null)
147 {
148 foreach (KeyValuePair<string, int> pair in data)
149 {
150 UpgradeDefinition upgrade = FindDefinition(pair.Key);
151 if (upgrade == null)
152 {
153 Debug.LogWarning($"UpgradeController: save has upgrade '{pair.Key}', but no asset in the list has that id. Skipped.");
154 continue;
155 }
156 SetLevel(upgrade.Id, Mathf.Clamp(pair.Value, 0, upgrade.MaxLevel));
157 }
158 }
159 UpgradesChanged?.Invoke();
160 }

175 private void SetLevel(string id, int level)
176 {
177 UpgradeLevel entry = owned.Find(e => e.Id == id);
178 if (entry == null) owned.Add(new UpgradeLevel { Id = id, Level = level });
179 else entry.Level = level;
180 }
181
182 // two assets with the same id would share one level, so warn about it early
183 private void CheckList()
184 {
185 HashSet<string> ids = new HashSet<string>();
186 foreach (UpgradeDefinition upgrade in upgrades)
187 {
188 if (upgrade == null) { Debug.LogWarning("UpgradeController: there's an empty slot in the Upgrades list."); continue; }
189 if (!ids.Add(upgrade.Id))
190 Debug.LogWarning($"UpgradeController: more than one upgrade has the id '{upgrade.Id}'. Give each asset its own id.");
191 }
192 if (upgrades.Count == 0)
193 Debug.LogWarning("UpgradeController: the Upgrades list is empty, so the shop will have nothing to sell.");
194 }
