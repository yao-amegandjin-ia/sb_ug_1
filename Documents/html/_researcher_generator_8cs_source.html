62 public void Generate()
63 {
64 if (!ValidateSettings()) return;
65
66 foreach (Researcher old in researchers)

68 researchers.Clear();
69

71 GameSeed.Set(seedOverride);

74
75 //Unique heads: shuffle all head indices, then deal one to each researcher.
76 List<int> heads = new List<int>();
77 for (int h = 0; h < headCount; h++) heads.Add(h);
78 Shuffle(heads, rng);

81 List<int> unused = new List<int>();
82 RefillPool(unused, rng);

98 int id = i + 1;

102 researchers.Add(r);

114 private List<int> DrawMethodologies(List<int> unused, Random rng)
115 {
116
117 List<int> unlocked = new List<int>();
118
119 while (unlocked.Count < 4)
120 {
121 // Start a new cycle (every methodology has now been used equally often).
122 if (unused.Count == 0)
123 RefillPool(unused, rng);
124
125 //finds the first index in unused, where it also is not in unlocked
126 int index = unused.FindIndex(m => !unlocked.Contains(m));
127
128 // Only duplicates of this researcher's methodologies are left; start a new cycle.
129 if (index < 0)
130 {
131 RefillPool(unused, rng);
132 index = unused.FindIndex(m => !unlocked.Contains(m));
133 }
134
135 unlocked.Add(unused[index]);
136 unused.RemoveAt(index);
137 }
138
139 return unlocked;
140 }
