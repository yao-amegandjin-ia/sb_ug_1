1// HireTab - the shop's hire tab. goes on the ShopUI object next to the ShopScreen.
2// this is what ResearcherHireScreen used to do: one ResearcherCard.uxml per researcher the FirmController says can be hired.
3// HIRE asks the FirmController to put them in a cubicle. the screen stays open and the list rebuilds without them.

5using Researchers;

16 public override void Fill(VisualElement list)
17 {
18 if (!Firm) return;

23 return;

39 {
40 VisualElement card = researcherCard.Instantiate();

42 SetText(card, "researcher-name", r.Name);
43 SetText(card, "quality", "QUALITY " + r.Quality + "/3");
44 SetText(card, "speed", "SPEED " + r.Speed + "/3");
45
46 // no pictures yet, so the portrait shows initials. once there are sprites:
47 // card.Q("portrait").style.backgroundImage = new StyleBackground(sprite);
48 // card.Q("portrait-initials").style.display = DisplayStyle.None;

50
51 new MethodologyTray(card.Q("methods")).Show(r.KnownMethodologyIds);

67 string initials = "";
