73 // Replaces the old Update(). The Client component calls this every frame with Time.deltaTime.
74 public void Tick(float deltaTime)
75 {
76 // patience only drops while the client hasn't been helped yet
77 // (opening the request panel does NOT pause it, per the GDD)
78 if (state == ClientState.InProcess || state == ClientState.Waiting)
79 {
80 currentPatience -= deltaTime;
81
82 if (currentPatience <= 0f)
83 {
84 currentPatience = 0f;
85 SetState(ClientState.LeftDissatisfied);
86 OnLeftDissatisfied?.Invoke(this);
87 }
88 }
89 }
