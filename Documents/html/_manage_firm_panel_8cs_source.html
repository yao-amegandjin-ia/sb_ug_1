58 void RefreshResearchers()
59 {
60 researcherList.Clear();
61 int count = 0;
62
63 ResearcherGenerator generator = FindAnyObjectByType<ResearcherGenerator>();
64 if (generator != null)
65 {
66 foreach (ResearcherData r in generator.Researchers)
67 {
68 if (!r.IsPurchased) continue;
69 string detail = "Cubicle " + r.Position + " · Speed " + r.Speed + " · Quality " + r.Quality + " · XP " + r.Experience;
70 if (r.CurrentState == ResearcherState.Busy)
71 detail += " · busy";
72 AddResearcherCard(r.Name, detail, r.KnownMethodologyIds);
73 count++;
74 }
75 }
76
77 // nobody is hired through the generator yet, so show the starter from the start screen
78 if (count == 0)
79 {
80 AddResearcherCard(startScreen.GetStarterName(), "Starting researcher", new int[] { startScreen.GetStarterMethod() });
81 }
82 }

110 void RefreshUpgrades()
111 {
112 upgradeList.Clear();
113
114 UpgradeController upgrades = FindAnyObjectByType<UpgradeController>();
115 if (upgrades == null) return;
116
117 foreach (UpgradeDefinition upgrade in upgrades.All)
118 {
119 if (upgrade == null) continue;
120 int level = upgrades.GetLevel(upgrade);
121
122 VisualElement card = new VisualElement();
123 card.AddToClassList("firm-card");
124
125 string levelText = level == 0 ? "Not bought yet" : "Level " + level + "/" + upgrade.MaxLevel;
126 Label nameLabel = new Label(upgrade.DisplayName + " (" + levelText + ")");
127 nameLabel.AddToClassList("firm-name");
128 Label desc = new Label(upgrade.Description);
129 desc.AddToClassList("firm-detail");
130
131 card.Add(nameLabel);
132 card.Add(desc);
133 upgradeList.Add(card);
134 }
135 }
